    const emailUser = process.env.EMAIL_USER

    if (emailUser && emailPass) {
      this.hasCredentials = true
      this.transporter = nodemailer.createTransport({
        service: 'gmail',
        auth: {
          user: emailUser,
          pass: emailPass,
        },
      })
    } else {
      this.hasCredentials = false
    }
